# 02 — Class Skills Data (클래스 스킬 데이터)

> 6계통 전 스킬 레퍼런스. 구현 시 이 문서를 기준으로 한다.

---

## 범례

| 항목 | 설명 |
|------|------|
| 등급 | 기본(1.0x, 0기), 초급(1.5x, 10기), 중급(2.0x, 25기), 상급(3.0x, 50기), 극의(5.0x, 전량) |
| 원소 | 목(木), 화(火), 토(土), 금(金), 수(水), 무(無) |
| 대상 | 단일적, 전체적, 자신, 단일아군, 전체아군, 랜덤적(N) |
| 유형 | 물리, 술법, 회복, 버프, 디버프, 상태이상, 패시브 |
| 회복 공식 | SPI x 배율 x (1 + 대상_SPI x 0.01) |
| 물리 데미지 | (물리공격 x 배율) - (대상 물리방어 x 0.4) |
| 술법 데미지 | (술법공격 x 배율) - (대상 술법방어 x 0.35) |

### 습득 조건 약어

- `T0 기본` = 해당 Tier 0 클래스 시작 시 자동 습득
- `Lv N` = 레벨 N 이상
- `T1:클래스명` = 해당 T1 클래스 전직 시 습득
- `스탯≥N` = 특정 스탯이 N 이상

## 구현 참고: 스킬 데이터 구조

```typescript
interface Skill {
  id: string;                    // 고유 ID: "kenshi_t0_kiri" 형태
  name: string;                  // 한글명: "베기"
  nameJp: string;                // 일본어명: "斬り"
  tier: number;                  // 0~6
  grade: 'basic' | 'beginner' | 'intermediate' | 'advanced' | 'ultimate';
  multiplier: number;            // 배율 (극의는 가변)
  kiCost: number;                // 기 소비 (극의: -1 = 전량)
  element: Element;              // 'wood'|'fire'|'earth'|'metal'|'water'|'none'
  target: TargetType;            // 'single_enemy'|'all_enemy'|'random_enemy'|'self'|'single_ally'|'all_ally'
  targetCount?: number;          // random_enemy일 때 횟수
  type: 'physical' | 'magical' | 'heal' | 'buff' | 'debuff' | 'status' | 'passive';
  effects: SkillEffect[];       // 추가 효과 배열
  classLineage: string;          // 계통: 'kenshi'|'miko'|'shinobi'|'yamabushi'|'shokunin'|'kemono'
  classId: string;               // 습득 클래스 ID
  learnCondition: LearnCondition; // 습득 조건
  description: string;           // 효과 설명
}

type Element = 'wood' | 'fire' | 'earth' | 'metal' | 'water' | 'none' | 'variable';

type TargetType =
  | 'single_enemy'
  | 'all_enemy'
  | 'random_enemy'
  | 'self'
  | 'single_ally'
  | 'all_ally'
  | 'all_ally_and_enemy';

interface SkillEffect {
  type: 'damage' | 'heal' | 'buff' | 'debuff' | 'status_ailment' | 'special';
  stat?: string;
  value?: number;
  duration?: number;
  chance?: number;
  statusId?: string;
}

interface LearnCondition {
  type: 'class_default' | 'level' | 'stat' | 'event';
  level?: number;
  stat?: { name: string; min: number };
  classId?: string;
}
```

> 이 TypeScript 인터페이스를 기반으로 스킬 JSON 데이터를 생성할 것.
> 등급 → grade 매핑: 기본=basic, 초급=beginner, 중급=intermediate, 상급=advanced, 극의=ultimate
